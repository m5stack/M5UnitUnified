168inline I2CPins i2cPins(const NessoPort nesso = NessoPort::PortB)
169{
170 const auto board = M5.getBoard();
171 if (board == m5::board_t::board_ArduinoNessoN1) {
172 if (nesso == NessoPort::PortB) {
173 return {static_cast<int8_t>(M5.getPin(m5::pin_name_t::port_b_out)),
174 static_cast<int8_t>(M5.getPin(m5::pin_name_t::port_b_in)), I2CPins::Backend::SoftwareI2C};
175 }
176 return {static_cast<int8_t>(M5.getPin(m5::pin_name_t::port_a_sda)),
177 static_cast<int8_t>(M5.getPin(m5::pin_name_t::port_a_scl)), I2CPins::Backend::Wire};
178 }
179 if (board == m5::board_t::board_M5NanoC6 || board == m5::board_t::board_M5NanoH2) {
180 return {static_cast<int8_t>(M5.getPin(m5::pin_name_t::ex_i2c_sda)),
181 static_cast<int8_t>(M5.getPin(m5::pin_name_t::ex_i2c_scl)), I2CPins::Backend::ExI2C};
182 }
183 // In_I2C and Ex_I2C on the same pins are one bus that M5.begin() already installed via In_I2C
184 // (Core Basic/Gray/Go/Fire, ToughC5, CoreMatrix, PaperDIY...). Opening a new Wire / native master bus
185 // on it would collide, so borrow M5.Ex_I2C instead (ExI2C backend -> i2cClass).
186 const int8_t ex_sda = M5.getPin(m5::pin_name_t::ex_i2c_sda);
187 const int8_t ex_scl = M5.getPin(m5::pin_name_t::ex_i2c_scl);
188 if (ex_sda >= 0 && ex_scl >= 0 && ex_sda == M5.getPin(m5::pin_name_t::in_i2c_sda) &&
189 ex_scl == M5.getPin(m5::pin_name_t::in_i2c_scl)) {
190 return {ex_sda, ex_scl, I2CPins::Backend::ExI2C};
191 }
192 return {static_cast<int8_t>(M5.getPin(m5::pin_name_t::port_a_sda)),
193 static_cast<int8_t>(M5.getPin(m5::pin_name_t::port_a_scl)), I2CPins::Backend::Wire};
194}

203inline GpioPinPair gpioPins(const GpioRole role = GpioRole::Both)
204{
205 int rx = (role == GpioRole::OutOnly) ? -1 : M5.getPin(m5::pin_name_t::port_b_in);
206 int tx = (role == GpioRole::InOnly) ? -1 : M5.getPin(m5::pin_name_t::port_b_out);
207 const bool need_rx_fb = (role != GpioRole::OutOnly && rx < 0);
208 const bool need_tx_fb = (role != GpioRole::InOnly && tx < 0);
209 bool fb = false;
210 if (need_rx_fb || need_tx_fb) {
211 fb = true;
212 rx = (role == GpioRole::OutOnly) ? -1 : M5.getPin(m5::pin_name_t::port_a_pin1);
213 tx = (role == GpioRole::InOnly) ? -1 : M5.getPin(m5::pin_name_t::port_a_pin2);
214 }
215 return {static_cast<int8_t>(rx), static_cast<int8_t>(tx), fb};
216}

363inline bool addI2C(UnitUnified& units, Component& unit, const uint32_t clock = 0,
364 const NessoPort nesso = NessoPort::PortB)
365{
366 // clock == 0: use the unit's own configured clock; clock > 0: override it. Either way the unit's
367 // component_config().clock is the single source of truth applied per transaction by every backend.
368 if (clock != 0) {
369 auto cfg = unit.component_config();
370 cfg.clock = clock;
371 unit.component_config(cfg);
372 }
373 const uint32_t eff_clock = unit.component_config().clock;
374 const auto p = i2cPins(nesso);
375 M5_LIB_LOGI("wiring: addI2C board=0x%02x nesso=%d backend=%d sda=%d scl=%d clock=%lu", (int)M5.getBoard(),
376 (int)nesso, (int)p.backend, (int)p.sda, (int)p.scl, (unsigned long)eff_clock);
377 switch (p.backend) {
378 case I2CPins::Backend::SoftwareI2C:
379 return i2cSoftware(units, unit, p.sda, p.scl);
380 case I2CPins::Backend::Wire:
381 return i2cWire(units, unit, Wire, p.sda, p.scl, eff_clock);
382 case I2CPins::Backend::ExI2C:
383 return i2cClass(units, unit, M5.Ex_I2C);
384 }
385 return false;
386}

447inline bool addUART(UnitUnified& units, Component& unit, const uint32_t baud = 115200,
448 const UartConfig config = UartConfig::Default)
449{
450 HardwareSerial& serial = defaultUartSerial();
451 const uint32_t cfg = toArduinoSerialConfig(config);
452 const auto p = uartPins();
453 if (p.fallback_a) {
454 const auto b = M5.getBoard();
455 if (b == m5::board_t::board_M5NanoC6 || b == m5::board_t::board_M5NanoH2) {
456 // M5.begin() calls Ex_I2C.setPort() but NOT begin(). m5gfx::i2c::release() is gated
457 // by the internal `initialized` flag, so release() alone is a no-op. Workaround:
458 // begin() flips initialized=true, then release() actually detaches + restores pins.
459 M5.Ex_I2C.begin();
460 M5.Ex_I2C.release();
461 }
462 }
463 M5_LIB_LOGI("wiring: UART rx=%d tx=%d baud=%lu config=0x%lx fallback_a=%d", (int)p.rx, (int)p.tx,
464 (unsigned long)baud, (unsigned long)cfg, (int)p.fallback_a);
465 serial.end();
466 serial.begin(baud, cfg, p.rx, p.tx);
467 return units.add(unit, serial);
468}

484inline bool addSPI(UnitUnified& units, Component& unit, const uint32_t clock_hz, const uint8_t mode = 0,
485 const uint8_t bit_order = 0)
486{
487 SPISettings settings{clock_hz, static_cast<uint8_t>((bit_order == 0) ? MSBFIRST : LSBFIRST), mode};
488 const auto p = spiPins();
489 M5_LIB_LOGI("wiring: addSPI sclk=%d miso=%d mosi=%d clock=%lu mode=%u bit_order=%u", (int)p.sclk, (int)p.miso,
490 (int)p.mosi, (unsigned long)clock_hz, (unsigned)mode, (unsigned)bit_order);
491 if (!SPI.bus()) {
492 SPI.begin(p.sclk, p.miso, p.mosi);
493 }
494 return spiBus(units, unit, SPI, settings);
495}

504inline bool addHatI2C(UnitUnified& units, Component& unit, const uint32_t clock = 0)
505{
506 const auto p = hatI2CPins();
507 if (p.sda < 0 || p.scl < 0) {
508 M5_LIB_LOGE("wiring: Hat I2C unsupported board=0x%02x", (int)M5.getBoard());
509 return false;
510 }
511 // clock == 0: use the unit's own configured clock; clock > 0: override it.
512 if (clock != 0) {
513 auto cfg = unit.component_config();
514 cfg.clock = clock;
515 unit.component_config(cfg);
516 }
517 const uint32_t eff_clock = unit.component_config().clock;
518#if M5_UNIT_UNIFIED_WIRING_HAS_WIRE1
519 TwoWire& wire = p.useWire1 ? Wire1 : Wire;
520#else
521 // Arduino-ESP32 declares Wire only (e.g. ESP32-C3); Wire1 is absent.
522 if (p.useWire1) {
523 M5_LIB_LOGE("wiring: addHatI2C NessoN1 Hat needs Wire1, but Wire1 is not available");
524 return false;
525 }
526 TwoWire& wire = Wire;
527#endif
528 M5_LIB_LOGI("wiring: addHatI2C board=0x%02x sda=%d scl=%d clock=%lu wire=%s", (int)M5.getBoard(), (int)p.sda,
529 (int)p.scl, (unsigned long)eff_clock, p.useWire1 ? "Wire1" : "Wire");
530 return i2cWire(units, unit, wire, p.sda, p.scl, eff_clock);
531}

648#if __has_include(<driver/i2c_master.h>)
650inline i2c_master_bus_handle_t ensureI2CBus(const i2c_port_t port, const gpio_num_t sda, const gpio_num_t scl,
651 const uint32_t /*clock - not used to key, single bus per pins*/)
652{
653 const uint32_t key = (static_cast<uint32_t>(static_cast<uint8_t>(port)) << 24) |
654 (static_cast<uint32_t>(static_cast<uint8_t>(sda)) << 16) |
655 (static_cast<uint32_t>(static_cast<uint8_t>(scl)) << 8);
656 static I2CCacheEntry cache[kI2CBusCacheSize]{};
657 static size_t count{};
658 for (size_t i = 0; i < count; ++i) {
659 if (cache[i].key == key) return cache[i].handle;
660 }
661 if (count >= kI2CBusCacheSize) {
662 M5_LIB_LOGE("wiring: I2C bus cache full (max %zu)", kI2CBusCacheSize);
663 return nullptr;
664 }
665 i2c_master_bus_config_t cfg{};
666 cfg.i2c_port = port;
667 cfg.sda_io_num = sda;
668 cfg.scl_io_num = scl;
669#if SOC_LP_I2C_SUPPORTED
670 if (port == LP_I2C_NUM_0) {
671#if ESP_IDF_VERSION >= ESP_IDF_VERSION_VAL(5, 3, 0)
672 // LP I2C uses RTC_FAST clock (union with clk_source, mutually exclusive).
673 cfg.lp_source_clk = LP_I2C_SCLK_DEFAULT;
674#else
675 // i2c_master_bus_config_t has no lp_source_clk before ESP-IDF 5.3; LP I2C cannot be opened
676 M5_LIB_LOGE("wiring: LP I2C needs ESP-IDF >= 5.3");
677 return nullptr;
678#endif
679 } else
680#endif
681 {
682 cfg.clk_source = I2C_CLK_SRC_DEFAULT;
683 }
684 cfg.glitch_ignore_cnt = 7;
685 cfg.flags.enable_internal_pullup = true;
686 if (i2c_new_master_bus(&cfg, &cache[count].handle) != ESP_OK) {
687 M5_LIB_LOGE("wiring: i2c_new_master_bus failed port=%d sda=%d scl=%d", (int)port, (int)sda, (int)scl);
688 return nullptr;
689 }
690 cache[count].key = key;
691 return cache[count++].handle;
692}
693#else // legacy driver/i2c.h (IDF 5.0 / 5.1)
695inline bool ensureI2CLegacyDriver(const i2c_port_t port, const gpio_num_t sda, const gpio_num_t scl,
696 const uint32_t clock)
697{
698 const uint32_t key = (static_cast<uint32_t>(static_cast<uint8_t>(port)) << 24) |
699 (static_cast<uint32_t>(static_cast<uint8_t>(sda)) << 16) |
700 (static_cast<uint32_t>(static_cast<uint8_t>(scl)) << 8);
701 static uint32_t cache[kI2CBusCacheSize]{};
702 static size_t count{};
703 for (size_t i = 0; i < count; ++i) {
704 if (cache[i] == key) return true;
705 }
706 if (count >= kI2CBusCacheSize) {
707 M5_LIB_LOGE("wiring: I2C bus cache full (max %zu)", kI2CBusCacheSize);
708 return false;
709 }
710 // i2c_driver_install returns ESP_FAIL for several distinct reasons in IDF 5.0/5.1 (already
711 // installed by another owner / OOM / slave-setup failure). Probe i2c_get_period to disambiguate:
712 // it writes high/low only when the driver object exists (p_i2c_obj != NULL), so high>0 && low>0
713 // means the port is genuinely installed and can be borrowed.
714 const esp_err_t install_err = i2c_driver_install(port, I2C_MODE_MASTER, 0, 0, 0);
715 if (install_err == ESP_OK) {
716 // Freshly installed by us: configure pins and timing.
717 i2c_config_t conf{};
718 conf.mode = I2C_MODE_MASTER;
719 conf.sda_io_num = sda;
720 conf.scl_io_num = scl;
721 conf.sda_pullup_en = true;
722 conf.scl_pullup_en = true;
723 conf.master.clk_speed = clock;
724 if (i2c_param_config(port, &conf) != ESP_OK) {
725 M5_LIB_LOGE("wiring: legacy i2c param_config failed port=%d sda=%d scl=%d", (int)port, (int)sda, (int)scl);
726 i2c_driver_delete(port);
727 return false;
728 }
729 M5_LIB_LOGI("wiring: legacy i2c INSTALLED(new) port=%d sda=%d scl=%d clock=%u", (int)port, (int)sda, (int)scl,
730 (unsigned)clock);
731 } else if (install_err == ESP_FAIL) {
732 // Already installed by another owner -> borrow it (keep the owner's pins/timing; do NOT
733 // re-run i2c_param_config and override them). Confirm the driver really exists first.
734 int high = 0, low = 0;
735 i2c_get_period(port, &high, &low);
736 if (!(high > 0 && low > 0)) {
737 M5_LIB_LOGE("wiring: legacy i2c driver install failed port=%d sda=%d scl=%d", (int)port, (int)sda,
738 (int)scl);
739 return false;
740 }
741 M5_LIB_LOGI("wiring: legacy i2c BORROW(already) port=%d sda=%d scl=%d high=%d low=%d", (int)port, (int)sda,
742 (int)scl, high, low);
743 } else {
744 M5_LIB_LOGE("wiring: legacy i2c driver install error=0x%x port=%d sda=%d scl=%d", (int)install_err, (int)port,
745 (int)sda, (int)scl);
746 return false;
747 }
748 cache[count++] = key;
749 return true;
750}

754inline uart_port_t ensureUARTPort(const uart_port_t port, const gpio_num_t rx, const gpio_num_t tx, const uint32_t baud,
755 const UartConfig config)
756{
757 const uint32_t key = (static_cast<uint32_t>(static_cast<uint8_t>(port)) << 24) |
758 (static_cast<uint32_t>(static_cast<uint8_t>(rx)) << 16) |
759 (static_cast<uint32_t>(static_cast<uint8_t>(tx)) << 8);
760 static UARTCacheEntry cache[kUARTPortCacheSize]{};
761 static size_t count{};
762 for (size_t i = 0; i < count; ++i) {
763 if (cache[i].key == key) return cache[i].port;
764 }
765 if (count >= kUARTPortCacheSize) {
766 M5_LIB_LOGE("wiring: UART port cache full (max %zu)", kUARTPortCacheSize);
767 return UART_NUM_MAX;
768 }
769 if (uart_is_driver_installed(port)) {
770 cache[count] = {key, port};
771 return cache[count++].port;
772 }
773 const auto p = toIdfUartParams(config);
774 uart_config_t uc{};
775 uc.baud_rate = static_cast<int>(baud);
776 uc.data_bits = p.word_length;
777 uc.parity = p.parity;
778 uc.stop_bits = p.stop_bits;
779 uc.flow_ctrl = UART_HW_FLOWCTRL_DISABLE;
780 uc.source_clk = UART_SCLK_DEFAULT;
781 if (uart_param_config(port, &uc) != ESP_OK ||
782 uart_set_pin(port, tx, rx, UART_PIN_NO_CHANGE, UART_PIN_NO_CHANGE) != ESP_OK ||
783 uart_driver_install(port, 256, 0, 0, nullptr, 0) != ESP_OK) {
784 M5_LIB_LOGE("wiring: uart setup failed port=%d", (int)port);
785 return UART_NUM_MAX;
786 }
787 cache[count] = {key, port};
788 return cache[count++].port;
789}

793inline spi_device_handle_t ensureSPIDevice(const spi_host_device_t host, const gpio_num_t mosi, const gpio_num_t miso,
794 const gpio_num_t sck, const uint32_t clock_hz, const uint8_t mode,
795 const uint8_t bit_order)
796{
797 static SPIHostEntry host_cache[kSPIHostCacheSize]{};
798 static SPIDevEntry dev_cache[kSPIDevCacheSize]{};
799 static size_t host_count{};
800 static size_t dev_count{};
801
802 bool host_init = false;
803 for (size_t i = 0; i < host_count; ++i) {
804 if (host_cache[i].host == host) {
805 host_init = host_cache[i].initialized;
806 break;
807 }
808 }
809 if (!host_init) {
810 spi_bus_config_t bc{};
811 bc.mosi_io_num = mosi;
812 bc.miso_io_num = miso;
813 bc.sclk_io_num = sck;
814 bc.quadwp_io_num = -1;
815 bc.quadhd_io_num = -1;
816 bc.max_transfer_sz = 4096;
817 if (spi_bus_initialize(host, &bc, SPI_DMA_CH_AUTO) != ESP_OK) {
818 M5_LIB_LOGE("wiring: spi_bus_initialize failed host=%d", (int)host);
819 return nullptr;
820 }
821 if (host_count < kSPIHostCacheSize) host_cache[host_count++] = {host, true};
822 }
823
824 const uint32_t key = (static_cast<uint32_t>(static_cast<uint8_t>(host)) << 24) |
825 (static_cast<uint32_t>(clock_hz / 1000U) << 8) | (static_cast<uint32_t>(mode) << 4) |
826 static_cast<uint32_t>(bit_order);
827 for (size_t i = 0; i < dev_count; ++i) {
828 if (dev_cache[i].key == key) return dev_cache[i].handle;
829 }
830 if (dev_count >= kSPIDevCacheSize) {
831 M5_LIB_LOGE("wiring: SPI device cache full (max %zu)", kSPIDevCacheSize);
832 return nullptr;
833 }
834 spi_device_interface_config_t dc{};
835 dc.clock_speed_hz = static_cast<int>(clock_hz);
836 dc.mode = mode;
837 dc.spics_io_num = GPIO_NUM_NC; // manual CS control by adapter
838 dc.queue_size = 1;
839 if (bit_order == 1) dc.flags |= SPI_DEVICE_BIT_LSBFIRST;
840 if (spi_bus_add_device(host, &dc, &dev_cache[dev_count].handle) != ESP_OK) {
841 M5_LIB_LOGE("wiring: spi_bus_add_device failed host=%d", (int)host);
842 return nullptr;
843 }
844 dev_cache[dev_count].key = key;
845 return dev_cache[dev_count++].handle;
846}

898inline bool addI2C(UnitUnified& units, Component& unit, const uint32_t clock = 0,
899 const NessoPort nesso = NessoPort::PortB)
900{
901 // clock == 0: use the unit's own configured clock; clock > 0: override it. component_config().clock
902 // is the single source of truth applied per transaction (Wire / I2C_Class / master device / legacy).
903 if (clock != 0) {
904 auto cfg = unit.component_config();
905 cfg.clock = clock;
906 unit.component_config(cfg);
907 }
908 const uint32_t eff_clock = unit.component_config().clock;
909 const auto pins = i2cPins(nesso);
910 const auto board = M5.getBoard();
911 M5_LIB_LOGI("wiring(ESP-IDF): addI2C board=0x%02x nesso=%d sda=%d scl=%d clock=%u backend=%d", (int)board,
912 (int)nesso, (int)pins.sda, (int)pins.scl, (unsigned)eff_clock, (int)pins.backend);
913 switch (pins.backend) {
914 case I2CPins::Backend::SoftwareI2C:
915#if defined(M5_HAL_HPP)
916 return i2cSoftware(units, unit, pins.sda, pins.scl);
917#else
918 M5_LIB_LOGE("wiring: SoftwareI2C backend requires M5HAL");
919 return false;
920#endif
921 case I2CPins::Backend::Wire:
922 if (board == m5::board_t::board_ArduinoNessoN1) {
923 // NessoN1 PortA (QWIIC) -> Wire == M5.In_I2C (NessoN1 maps Wire to In_I2C).
924 // M5Unified already holds I2C_NUM_0 here, so borrow instead of installing.
925 return i2cClass(units, unit, M5.In_I2C);
926 } else {
927 // Stick / S3 etc: M5Unified does not call Wire.begin(), so install ourselves
928 // (Core is routed to ExI2C by i2cPins() and never reaches this branch)
929#if __has_include(<driver/i2c_master.h>)
930 auto bus = i2cBusHandle(I2C_NUM_0, (gpio_num_t)pins.sda, (gpio_num_t)pins.scl, eff_clock);
931 if (!bus) return false;
932 return units.add(unit, bus);
933#else // legacy driver (IDF 5.0 / 5.1): install legacy driver, then add by port
934 if (!detail::ensureI2CLegacyDriver(I2C_NUM_0, (gpio_num_t)pins.sda, (gpio_num_t)pins.scl, eff_clock)) {
935 return false;
936 }
937 return units.add(unit, I2C_NUM_0, (gpio_num_t)pins.sda, (gpio_num_t)pins.scl);
938#endif
939 }
940 case I2CPins::Backend::ExI2C:
941 // Core / NanoC6 / NanoH2: M5Unified manages Ex_I2C (Core shares it with In_I2C) -> borrow
942 return i2cClass(units, unit, M5.Ex_I2C);
943 }
944 return false;
945}
